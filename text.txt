    // Render the countdown day/hour/minute/second display
    function updateCdDisplay() {
      // Get the display container
      const disp = document.getElementById("cdDisplay");
      // Get the motivation card
      const mot = document.getElementById("cdMotivation");
      // Return silently if containers are missing
      if (!disp || !mot) return;
      // Compute the time difference to the next official start
      const diff = getExamDiff();
      // Format the countdown boxes
      disp.innerHTML = [
        ["Days", Math.max(0, Math.floor(diff / 86400000))],
        ["Hours", Math.max(0, Math.floor(diff / 3600000) % 24)],
        ["Minutes", Math.max(0, Math.floor(diff / 60000) % 60)],
        ["Seconds", Math.max(0, Math.floor(diff / 1000) % 60)]
      ].map(u =>
        `<div class="count-box"><div class="count-val" data-cd="${u[0].toLowerCase()}">${u[1]}</div><div class="count-label">${u[0]}</div></div>`
      ).join("");
      // Set the motivational message depending on time remaining
      const days = Math.floor(diff / 86400000);
      // Choose a message based on proximity to the exam
      const msg = days <= 0
        ? "Exam day is here or has passed. Stay calm, trust your preparation and give your best!"
        : days <= 7
          ? "Final week! Focus on revision summaries, weak topics and timed practice."
          : "You have time. Stay consistent, one question at a time, and your results will rise.";
      // Render the exam name and motivational message
      mot.innerHTML = `<span class="pill">${Exams[App.exam].name}</span><h3 style="margin-top:10px;">${msg}</h3>`;

    // Calculate the millisecond difference to the next official exam start
    function getExamDiff() {
      // Get the start date of the selected exam
      const start = new Date(Exams[App.exam].start).getTime();
      // Get the current time
      const now = Date.now();
      // If the start has passed, target the end date instead
      const target = start > now ? start : new Date(Exams[App.exam].end).getTime();
      // Return the difference in milliseconds
      return target - now;
    }

      App.cdTimer = setInterval(updateCdDisplay, 1000);
    }

    // Filter and render the current question list
    function renderQuestionList() {
      // Get the list container
      const list = document.getElementById("pqList");
      // Return silently if the container is missing
      if (!list) return;
      // Filter the question bank by current filters
      let questions = QuestionBank.filter(q =>
        q.exam === App.exam &&
        q.subject === App.subject &&
        q.year === App.year
      );
      // If no subject-specific questions, show all subjects for the exam and year
      if (questions.length === 0) {
        questions = QuestionBank.filter(q => q.exam === App.exam && q.year === App.year);
      }
      // If still empty, show a helpful empty state
      if (questions.length === 0) {
        list.innerHTML = `<div class="card"><p>No questions found for <strong>${App.exam} ${App.subject} ${App.year}</strong>. Try clearing a filter or choosing another subject. Authentic past questions are added by admins and clearly labelled.</p></div>`;
        // Stop here
        return;
      }
      // Render each question card
      list.innerHTML = questions.map((q, idx) => `
        <div class="question" data-qid="${idx}" data-answer="${q.answer}">
          <!-- Question meta pills -->
          <div style="display:flex;gap:8px;flex-wrap:wrap;margin-bottom:8px;">
            <span class="pill">${q.exam} &middot; ${q.subject} &middot; ${q.year}</span>

          </div>
          <!-- Question prompt -->
          <div class="qtext">${q.q}</div>

            ${q.choices.map((c, i) => `<div class="choice" data-choice="${i}"><strong>${"ABCD"[i]}.</strong> <span>${c}</span></div>`).join("")}

        </div>

          // Get the parent question card
          const card = el.closest(".question");
          // Clear previous highlights in this card
          card.querySelectorAll(".choice").forEach(c => c.classList.remove("correct", "wrong"));

      // Attach check answer button handlers
      list.querySelectorAll("[data-reveal]").forEach(btn => {
        // Reveal the correct answer and explanation
        btn.addEventListener("click", () => {

          const card = btn.closest(".question");
          const correctIdx = parseInt(card.dataset.answer, 10);
          // Highlight the correct choice
          const correctEl = card.querySelector(`.choice[data-choice="${correctIdx}"]`);
          if (correctEl) correctEl.classList.add("correct");
          // Find the question object for the explanation
          let questions = QuestionBank.filter(q => q.exam === App.exam && q.subject === App.subject && q.year === App.year);
          if (questions.length === 0) questions = QuestionBank.filter(q => q.exam === App.exam && q.year === App.year);

        btn.addEventListener("click", () => {
          // Build a unique bookmark id

          // Persist the change
          saveData();
          // Show a toast
          toast(App.bookmarks.includes(id) ? "Question bookmarked for review." : "Bookmark removed.");
        });
      });
    }

    // Start a timed mock session
    function startTimedMock() {
      // Get the timer container and display
      const wrap = document.getElementById("pqTimerWrap");
      const disp = document.getElementById("pqTimer");
      // Show the timer panel
      wrap.classList.remove("hidden");
      // Set a 10 minute mock duration in seconds
      let remaining = 600;
      // Update the display helper
      function tick() {
        // Format as mm:ss
        const m = Math.floor(remaining / 60).toString().padStart(2, "0");
        const s = (remaining % 60).toString().padStart(2, "0");
        disp.textContent = m + ":" + s;
        // If time is up, auto submit
        if (remaining <= 0) submitMock();
        // Decrement
        remaining--;
      }
      // Clear any previous mock timer
      if (App.pqMockTimer) clearInterval(App.pqMockTimer);
      // Run the tick once immediately
      tick();
      // Run every second
      App.pqMockTimer = setInterval(tick, 1000);
      // Show a toast
      toast("Timed mock started. You have 10 minutes.");
    }

    // Submit the timed mock and grade answers
    function submitMock() {
      // Clear the mock timer
      if (App.pqMockTimer) clearInterval(App.pqMockTimer);
      // Get the results container
      const res = document.getElementById("pqResults");
      // Get all answered questions
      const cards = document.querySelectorAll("#pqList .question");
      // Count correct and total
      let correct = 0;
      let total = 0;
      // Grade each card
      cards.forEach(card => {
        // Find the selected choice
        const selected = card.querySelector(".choice.selected-choice");
        // Mark selected as answered
        total++;
        // Compare to the correct answer
        if (selected) {
          const selIdx = selected.dataset.choice;
          // Highlight correct and incorrect
          card.querySelectorAll(".choice").forEach(c => c.classList.remove("correct", "wrong"));
          const correctEl = card.querySelector(`.choice[data-choice="${card.dataset.answer}"]`);
          if (correctEl) correctEl.classList.add("correct");
          // Mark the wrong one if incorrect
          if (selIdx !== card.dataset.answer) selected.classList.add("wrong");
          // Increment correct if right
          if (selIdx === card.dataset.answer) correct++;
        }
      });
      // Compute percentage
      const pct = total ? Math.round((correct / total) * 100) : 0;
      // Update stats
      App.stats.attempted += total;
      App.stats.correct += correct;
      App.stats.sessions++;
      // Persist stats
      saveData();
      // Render the results panel
      res.classList.remove("hidden");
      res.innerHTML = `
        <h3>Mock Results</h3>
        <div class="grid grid-3">
          <div class="card"><div class="stat-num">${correct}/${total}</div><div style="color:var(--muted);font-size:0.8rem;">Correct</div></div>
          <div class="card"><div class="stat-num">${pct}%</div><div style="color:var(--muted);font-size:0.8rem;">Score</div></div>
          <div class="card"><div class="stat-num">${pct >= 70 ? "Strong" : "Keep Practising"}</div><div style="color:var(--muted);font-size:0.8rem;">Status</div></div>
        </div>
        <p style="margin-top:10px;">Review the highlighted choices above. Correct answers are marked green; your incorrect picks are marked red.</p>
      `;
      // Show a toast
      toast(`Mock submitted. You scored ${pct}%.`);
      // Scroll to the results panel
      res.scrollIntoView({ behavior: "smooth" });
    }

    // Initialize the Pomodoro study timer
    function initPomodoro() {
      // Get the timer display and buttons
      const disp = document.getElementById("pomTimer");
      const toggle = document.getElementById("pomToggle");
      const reset = document.getElementById("pomReset");
      const mode = document.getElementById("pomMode");
      // Default focus duration in seconds
      let focus = 25 * 60;
      // Break duration
      let breakTime = 5 * 60;
      // Current mode
      let isFocus = true;
      // Remaining seconds
      let remaining = focus;
      // Timer interval id
      let interval = null;
      // Rendering function for the display
      function render() {
        // Format as mm:ss
        const m = Math.floor(remaining / 60).toString().padStart(2, "0");
        const s = (remaining % 60).toString().padStart(2, "0");
        disp.textContent = m + ":" + s;
      }
      // Start / pause the timer
      toggle.addEventListener("click", () => {
        // If already running, pause
        if (interval) {
          clearInterval(interval);
          interval = null;
          toggle.textContent = "Start";
          // Stop here
          return;
        }
        // Start counting down
        interval = setInterval(() => {
          // Decrement
          remaining--;
          // Render the display
          render();
          // When zero, auto switch mode
          if (remaining <= 0) {
            // Clear the interval
            clearInterval(interval);
            interval = null;
            toggle.textContent = "Start";
            // Switch mode automatically
            switchMode();
          }
        }, 1000);
        // Update the button label
        toggle.textContent = "Pause";
        // Show a toast
        toast(isFocus ? "Focus session started. Stay on task!" : "Break started. Rest well.");
      });
      // Reset the timer
      reset.addEventListener("click", () => {
        // Clear any running interval
        if (interval) clearInterval(interval);
        interval = null;
        toggle.textContent = "Start";
        // Reset remaining to the current mode duration
        remaining = isFocus ? focus : breakTime;
        // Render
        render();
        toast("Timer reset.");
      });
      // Switch between focus and break modes
      function switchMode() {
        // Flip the mode
        isFocus = !isFocus;
        // Set the remaining to the new mode
        remaining = isFocus ? focus : breakTime;
        // Update the mode button label
        mode.textContent = isFocus ? "Break Mode" : "Focus Mode";
        // Render
        render();
      }
      // Toggle the mode button
      mode.addEventListener("click", () => {
        // Switch the mode
        switchMode();
      });
      // Initial render
      render();
    }

    // Compute the strongest and weakest subject from saved data
    function computeSubjectMastery() {
      // Get the elements
      const strongEl = document.getElementById("dashStrong");
      const weakEl = document.getElementById("dashWeak");
      const strongFill = document.getElementById("strongFill");
      const weakFill = document.getElementById("weakFill");
      // Return silently if missing
      if (!strongEl) return;
      // Load the study data
      const data = loadStudyData();
      // If no mastery data, keep default messages
      if (!data.mastery || data.mastery.length === 0) return;
      // Find the max and min mastery
      const max = data.mastery.reduce((a, b) => a.score > b.score ? a : b);
      const min = data.mastery.reduce((a, b) => a.score < b.score ? a : b);
      // Display the strongest subject
      strongEl.textContent = max.subject + " (" + Math.round(max.score * 100) + "% mastery)";
      strongFill.style.width = Math.round(max.score * 100) + "%";
      // Display the weakest subject
      weakEl.textContent = min.subject + " (" + Math.round(min.score * 100) + "% mastery)";
      weakFill.style.width = Math.round(min.score * 100) + "%";
    }

    // Bind events for the auth page
    function bindAuthEvents() {
      // Continue as guest
      const guestBtn = document.getElementById("guestBtn");
      if (guestBtn) {
        guestBtn.addEventListener("click", () => {
          // Clear any current user
          App.user = null;
          // Navigate to the dashboard
          navigate("dashboard");
          // Show a toast
          toast("You are now in guest mode. Progress is saved on this device.");
        });
      }
      // Register button
      const regBtn = document.getElementById("regBtn");
      if (regBtn) {
        regBtn.addEventListener("click", () => {
          // Read the registration fields
          const name = document.getElementById("regName").value.trim();
          const email = document.getElementById("regEmail").value.trim();
          const pass = document.getElementById("regPass").value;
          // Validate the fields
          if (!name || !email || pass.length < 6) {
            toast("Please fill all fields with a valid email and a password of 6+ characters.");
            // Stop here
            return;
          }
          // Load existing users
          const users = loadUsers();
          // Check for duplicate email
          if (users.some(u => u.email === email)) {
            toast("An account with that email already exists. Please log in.");
            // Stop here
            return;
          }
          // Create the new user object
          const newUser = { name, email, password: btoa(pass), createdAt: new Date().toLocaleDateString() };
          // Push to the users list
          users.push(newUser);
          // Persist the users
          localStorage.setItem("moraless_users", JSON.stringify(users));
          // Set the current user
          App.user = { name, email };
          // Update the auth button label
          document.getElementById("authBtn").textContent = "Signed In";
          // Show a toast
          toast("Account created. Welcome, " + name + "!");
          // Navigate to the dashboard
          navigate("dashboard");
        });
      }
      // Log in button
      const logBtn = document.getElementById("logBtn");
      if (logBtn) {
        logBtn.addEventListener("click", () => {
          // Read the login fields
          const email = document.getElementById("logEmail").value.trim();
          const pass = document.getElementById("logPass").value;
          // Load existing users
          const users = loadUsers();
          // Find the matching user
          const found = users.find(u => u.email === email && u.password === btoa(pass));
          // Validate the credentials
          if (!found) {
            toast("Invalid email or password.");
            // Stop here
            return;
          }
          // Set the current user
          App.user = { name: found.name, email: found.email };
          // Update the auth button label
          document.getElementById("authBtn").textContent = "Signed In";
          // Show a toast
          toast("Welcome back, " + found.name + "!");
          // Navigate to the dashboard
          navigate("dashboard");
        });
      }
      // Log out button
      const logOut = document.getElementById("logOut");
      if (logOut) {
        logOut.addEventListener("click", () => {
          // Clear the current user
          App.user = null;
          // Update the auth button label
          document.getElementById("authBtn").textContent = "Sign In";
          // Show a toast
          toast("You have been logged out.");
          // Re-render the auth page
          renderAuth();
        });
      }
    }

    // Initialise the animated star field background
    function initStars() {
      // Get the canvas and its context
      const canvas = document.getElementById("stars");
      const ctx = canvas.getContext("2d");
      // Store the star array
      let stars = [];
      // Resize the canvas to the viewport
      function resize() {
        // Set the canvas width to the window width
        canvas.width = window.innerWidth;
        // Set the canvas height to the window height
        canvas.height = window.innerHeight;
      }
      // Create the star objects
      function createStars() {
        // Clear the star array
        stars = [];
        // Determine the number of stars based on screen size
        const count = Math.floor((canvas.width * canvas.height) / 6000);
        // Build the star objects
        for (let i = 0; i < count; i++) {
          // Push a star with random position, speed and radius
          stars.push({
            x: Math.random() * canvas.width,
            y: Math.random() * canvas.height,
            r: Math.random() * 1.5 + 0.3,
            speed: Math.random() * 0.4 + 0.1,
            twinkle: Math.random() * Math.PI
          });
        }
      }
      // Draw and animate the stars
      function draw() {
        // Clear the canvas
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        // Loop through every star
        stars.forEach((s, i) => {
          // Move the star downward slowly
          s.y += s.speed;
          // Animate the twinkle
          s.twinkle += 0.03;
          // If the star moves off the bottom, reset it to the top
          if (s.y > canvas.height) s.y = 0;
          // Compute the twinkle alpha
          const alpha = 0.4 + Math.sin(s.twinkle) * 0.4;
          // Set the star color with alpha
          ctx.fillStyle = "rgba(200, 210, 255, " + alpha + ")";
          // Draw the star as a small circle
          ctx.beginPath();
          ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
          ctx.fill();
        });
        // Request the next animation frame
        requestAnimationFrame(draw);
      }
      // Initialise on load
      resize();
      createStars();
      draw();
      // Re-init on window resize
      window.addEventListener("resize", () => {
        resize();
        createStars();
      });
    }
